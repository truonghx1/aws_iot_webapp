📋 cURL Commands (Copy để test)

# Bật đèn (GET)
curl -X GET "https://YOUR-API-URL/prod/led?action=on"

# Tắt đèn (GET)  
curl -X GET "https://YOUR-API-URL/prod/led?action=off"

# Gửi lệnh tùy chỉnh (POST)
curl -X POST "https://YOUR-API-URL/prod/command" \
  -H "Content-Type: application/json" \
  -d '{"command":"LED_ON","data":{"brightness":50}}'
